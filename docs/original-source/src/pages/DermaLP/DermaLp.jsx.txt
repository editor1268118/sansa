function CountUp({ end }) {
  const [count, setCount] = useState(0);
  const { ref, inView } = useInView({
    triggerOnce: true,
    threshold: 0.1,
  });

  useEffect(() => {
    if (inView) {
      animateCount(0, end, 2000, setCount);
    }
  }, [inView, end]);

  return (
    <h2 style={{ color: "#fff" }} ref={ref}>
      {count}+
    </h2>
  );
}
